REM Author: NocX
REM Description: Turns on Firewall with powershell and enables real time protection using the Windows Defender GUI
REM Version: 1.0
REM Date: 17/01/2025
REM Category: Security
REM ------------------------------------
REM This script:
REM 1 Opens PowerShell (Admin)
REM 2 Enables Windows Firewall
REM 3 Opens Windows Security GUI settings
REM 4 Navigates and enables Real time Protection
REM 5 Closes everything
REM ------------------------------------

DEFAULT_DELAY 200

REM ---------- Open PowerShell as Admin ----------
DELAY 500
GUI r
DELAY 500
STRING powershell Start-Process powershell -Verb runAs
ENTER
DELAY 800
LEFTARROW
ENTER
DELAY 500

REM ---------- Enable Windows Firewall ----------
STRING netsh advfirewall set currentprofile state on
ENTER
DELAY 20
STRING exit
ENTER
DELAY 20

REM ---------- Open Windows Security ----------
REM Open the Start Menu
CONTROL ESCAPE
DELAY 100
STRING Sicurezza di Windows
ENTER
DELAY 900

REM ---------- Navigate to Real time protection ----------
ENTER
DELAY 1
TAB
DELAY 1
TAB
DELAY 1
TAB
DELAY 1
TAB
DELAY 1
ENTER
DELAY 500

REM ---------- admin permissions ----------
LEFTARROW
DELAY 50
ENTER
DELAY 100

REM ---------- Close window ----------
ALT F4
